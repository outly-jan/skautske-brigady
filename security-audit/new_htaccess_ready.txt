# --- Blokace wp-login.php utoku ---
<IfModule mod_rewrite.c>
    RewriteEngine On

    # Blokace spamu pres lostpassword (uz na webu)
    RewriteCond %{QUERY_STRING} action=lostpassword [NC]
    RewriteRule ^wp-login\.php$ - [F,L]

    # Blokace otevrene registrace
    RewriteCond %{QUERY_STRING} action=register [NC]
    RewriteRule ^wp-login\.php$ - [F,L]
</IfModule>

    RewriteRule .* - [F,L]
</IfModule>
